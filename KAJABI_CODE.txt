    // Generate token
    const response = await fetch(

      {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' }

  } catch (error) {
    console.error('Error loading chatbot:', error);

      '<div style="padding: 40px; text-align: center; color: #666;">' +
